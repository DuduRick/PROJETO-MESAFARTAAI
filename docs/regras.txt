"""
MESAFARTAI - Módulo de Processamento de Linguagem Natural (NLU)
Arquivo: src/nlu.py
Descrição: Pipeline com TF-IDF, Random Forest e Trava de Segurança por Threshold.
"""

import numpy as np
from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.ensemble import RandomForestClassifier

# 1. BASE DE DADOS DE TREINO (Dataset de Intenções)
frases_treino = [
    # Intenção: cadastrar_doacao
    "Quero doar 10 kg de arroz",
    "Posso contribuir com 15 litros de leite",
    "Tenho 30 kg de banana para doar",
    "Tenho 20 caixas de tomate para doar hoje",
    "Gostaria de oferecer refeições prontas para abrigo",
    "Tenho sobra de alimentos do meu restaurante",
    "Posso entregar 5 kg de feijão perto da validade",
    "Queremos cadastrar uma doação de legumes",
    
    # Intenção: solicitar_alimentos
    "Nossa ONG precisa de doações de alimentos",
    "Estamos buscando marmitas e cestas básicas urgentemente",
    "Quais doações estão disponíveis para retirada na região?",
    "Preciso de suprimentos para o abrigo comunitário",
    "Como faço para solicitar comida para nossa cozinha social?",
    
    # Intenção: consultar_status
    "Onde está o meu pedido de doação?",
    "Gostaria de ver o histórico de matches das ONGs",
    "Qual o status do voluntário que vai retirar os alimentos?",
    "Como consultar se a doação já foi entregue?",
    
    # Intenção: fora_de_escopo (Frases aleatórias / Ruídos)
    "Qual é a previsão do tempo para hoje?",
    "Gosto de assistir filmes de ficção científica",
    "Quem venceu o jogo de futebol ontem?",
    "Como faço para trocar a senha da minha conta bancária?"
]

intencoes_treino = [
    "cadastrar_doacao", "cadastrar_doacao", "cadastrar_doacao", 
    "cadastrar_doacao", "cadastrar_doacao", "cadastrar_doacao", "cadastrar_doacao", "cadastrar_doacao",
    
    "solicitar_alimentos", "solicitar_alimentos", "solicitar_alimentos", 
    "solicitar_alimentos", "solicitar_alimentos",
    
    "consultar_status", "consultar_status", "consultar_status", "consultar_status",
    
    "fora_de_escopo", "fora_de_escopo", "fora_de_escopo", "fora_de_escopo"
]

class MotorNLU:
    def __init__(self, limiar_threshold=0.60):
        self.limiar_threshold = limiar_threshold
        self.vectorizer = TfidfVectorizer(ngram_range=(1, 2))
        self.classifier = RandomForestClassifier(n_estimators=80, random_state=42)
        self._treinar_modelo()

    def _treinar_modelo(self):
        """Vetoriza os textos e treina o classificador Random Forest."""
        X_vec = self.vectorizer.fit_transform(frases_treino)
        self.classifier.fit(X_vec, intencoes_treino)
        print(" [NLU] Modelo Random Forest e Vetorizador TF-IDF treinados com sucesso!")

    def processar_mensagem(self, texto_usuario):
        """
        Processa uma mensagem do usuário e retorna a intenção, 
        o nível de confiança e a decisão de Fallback.
        """
        # 1. Vetorização da entrada
        vetor_texto = self.vectorizer.transform([texto_usuario])
        
        # 2. Predição de probabilidade via Random Forest
        probabilidades = self.classifier.predict_proba(vetor_texto)[0]
        max_confianca = float(np.max(probabilidades))
        indice_classe = int(np.argmax(probabilidades))
        intencao_predita = self.classifier.classes_[indice_classe]
        
        # 3. Trava de Segurança por Threshold
        if max_confianca < self.limiar_threshold or intencao_predita == "fora_de_escopo":
            return {
                "intencao": "fallback",
                "confianca": max_confianca,
                "resposta": (
                    "Desculpe, não consegui entender com clareza o seu pedido. \n"
                    "Você pode refazer a mensagem informando se deseja **doar alimentos** "
                    "ou **solicitar doações para uma ONG**?"
                )
            }
        
        return {
            "intencao": intencao_predita,
            "confianca": max_confianca,
            "resposta": f"Intenção '{intencao_predita}' identificada com {max_confianca*100:.1f}% de confiança."
        }

# --- TESTE DO MÓDULO EM TEMPO DE EXECUÇÃO ---
if __name__ == "__main__":
    nlu = MotorNLU(limiar_threshold=0.60)
    
    # Bateria de testes automatizados
    frases_teste = [
        "Tenho 30 kg de banana para doar",
        "Nossa instituição precisa de mantimentos",
        "Qual é a receita de bolo de cenoura?", # Frase fora de escopo
        "coisa comida doar la" # Frase confusa / ambígua
    ]
    
    print("\n--- BATERIA DE TESTES DE ESTRESSE NLU ---")
    for frase in frases_teste:
        resultado = nlu.processar_mensagem(frase)
        print(f"\nEntrada: '{frase}'")
        print(f"Intenção: {resultado['intencao']} | Confiança: {resultado['confianca']:.2f}")
        print(f"Retorno: {resultado['resposta']}")

"""
MESAFARTAI - Módulo de Extração de Entidades Nomeadas (Regex)
Arquivo: src/entities.py
"""

import re
from datetime import datetime, timedelta

class ExtratorEntidades:
    def __init__(self):
        # Padrão Regex para extrair números associados a unidades de peso
        self.padrao_quantidade = r'(\d+(?:[\.,]\d+)?)\s*(?:kg|quilos|kilos|caixas|sacas)'
        self.vocabulario_alimentos = [
            "arroz", "feijão", "feijao", "tomate", "banana", "leite", 
            "marmita", "marmitas", "pão", "pao", "legumes", "frutas", "carne"
        ]

    def extrair_quantidade(self, texto):
        match = re.search(self.padrao_quantidade, texto, re.IGNORECASE)
        if match:
            valor_str = match.group(1).replace(',', '.')
            return float(valor_str)
        return 1.0  # Valor padrão (1 kg) se não informado

    def extrair_alimento(self, texto):
        texto_lower = texto.lower()
        for item in self.vocabulario_alimentos:
            if item in texto_lower:
                return item.capitalize()
        return "Alimentos Diversos"

    def extrair_validade(self, texto):
        texto_lower = texto.lower()
        hoje = datetime.now()
        
        if "hoje" in texto_lower:
            return hoje.strftime("%Y-%m-%d")
        elif "amanhã" in texto_lower or "amanha" in texto_lower:
            return (hoje + timedelta(days=1)).strftime("%Y-%m-%d")
        else:
            return (hoje + timedelta(days=3)).strftime("%Y-%m-%d")

    def processar_texto(self, texto):
        return {
            "alimento": self.extrair_alimento(texto),
            "quantidade_kg": self.extrair_quantidade(texto),
            "data_validade": self.extrair_validade(texto)
        }

if __name__ == "__main__":
    extrator = ExtratorEntidades()
    frase_teste = "Tenho 15.5 kg de tomate para doar hoje até o final do dia"
    resultado = extrator.processar_texto(frase_teste)
    print("--- TESTE DE EXTRAÇÃO REGEX ---")
    print(f"Frase: '{frase_teste}'")
    print(f"Entidades Extraídas: {resultado}")

"""
MESAFARTAI - Motor de Matchmaking Logístico via KNN
Arquivo: src/matchmaking.py
Descrição: Localiza a ONG geograficamente mais próxima no espaço vetorial 2D.
"""

import sqlite3
import numpy as np
from sklearn.neighbors import NearestNeighbors


class MotorMatchmaking:
    def __init__(self, db_path="mesafartai.db"):
        self.db_path = db_path
        self._garantir_tabela_e_seeds()

    def _garantir_tabela_e_seeds(self):
        """Garante que a tabela 'usuarios' existe e possui dados de teste (ONGs)."""
        conn = sqlite3.connect(self.db_path)
        cursor = conn.cursor()

        # Criar a tabela caso não exista
        cursor.execute("""
            CREATE TABLE IF NOT EXISTS usuarios (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                nome TEXT NOT NULL,
                tipo TEXT NOT NULL,
                latitude REAL NOT NULL,
                longitude REAL NOT NULL
            )
        """)

        # Verificar se existem ONGs cadastradas
        cursor.execute("SELECT COUNT(*) FROM usuarios WHERE tipo = 'ONG'")
        total_ongs = cursor.fetchone()[0]

        # Se não houver ONGs, insere dados fictícios para teste
        if total_ongs == 0:
            ongs_seed = [
                ("ONG Prato Quente - Centro", "ONG", -23.612, -46.781),
                ("Abrigo Esperança - Zona Sul", "ONG", -23.630, -46.800),
                ("Instituto Fartura - Zona Oeste", "ONG", -23.590, -46.750)
            ]
            cursor.executemany("""
                INSERT INTO usuarios (nome, tipo, latitude, longitude)
                VALUES (?, ?, ?, ?)
            """, ongs_seed)
            conn.commit()

        conn.close()

    def buscar_ongs(self):
        """Busca todas as ONGs ativas cadastradas no banco de dados SQLite."""
        conn = sqlite3.connect(self.db_path)
        cursor = conn.cursor()
        cursor.execute("SELECT id, nome, latitude, longitude FROM usuarios WHERE tipo = 'ONG'")
        ongs = cursor.fetchall()
        conn.close()
        return ongs

    def encontrar_ong_mais_proxima(self, lat_doador, long_doador):
        """Aplica o algoritmo KNN para encontrar o vizinho mais próximo."""
        ongs = self.buscar_ongs()
        if not ongs:
            return None

        # Monta a matriz de coordenadas das ONGs: [[lat, long], ...]
        coords_ongs = np.array([[ong[2], ong[3]] for ong in ongs])

        # Treina o modelo KNN com distância Euclidiana (1 vizinho)
        knn = NearestNeighbors(n_neighbors=1, metric='euclidean')
        knn.fit(coords_ongs)

        # Coordenada do Doador
        ponto_doador = np.array([[lat_doador, long_doador]])

        # Calcula a menor distância
        distancias, indices = knn.kneighbors(ponto_doador)

        idx_ong_proxima = indices[0][0]
        distancia_calculada = distancias[0][0]
        ong_selecionada = ongs[idx_ong_proxima]

        return {
            "ong_id": ong_selecionada[0],
            "ong_nome": ong_selecionada[1],
            "distancia_estimada": round(float(distancia_calculada * 111), 2)  # Conversão aprox. em KM
        }


if __name__ == "__main__":
    motor = MotorMatchmaking()
    # Coordenada de teste (Ex: Supermercado/Doador)
    lat_teste, long_teste = -23.615, -46.785
    match = motor.encontrar_ong_mais_proxima(lat_teste, long_teste)

    print("--- TESTE DE MATCHMAKING LOGÍSTICO (KNN) ---")
    if match:
        print(f"ONG Encontrada: {match['ong_nome']} (ID: {match['ong_id']})")
        print(f"Distância Estimada: {match['distancia_estimada']} km")
    else:
        print("Nenhuma ONG cadastrada para realizar o match.")
